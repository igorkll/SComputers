a sign from scrap mechanic drilling thunder, only controlled from a computer
component name - smartsign


methods:

  • smartsign.setText(string)
  • smartsign.getText():string
  • smartsign.setTheme(num:number (1-8))
  • smartsign.getTheme():number
  • smartsign.getMaxTextLen():number

example:

            local smartsign = getComponent("smartsign")

local text = "hello, world!"

local currentTheme = 1
local maxTheme = 8

local maxText = smartsign.getMaxTextLen()
local textOffset = 0

function onTick()
    smartsign.setText(text:sub(1 + textOffset, maxText + textOffset))
    smartsign.setTheme((currentTheme % maxTheme) + 1)

    textOffset = textOffset + 1
    if textOffset >= maxText then
        textOffset = 0
        currentTheme = currentTheme + 1
    end
end

function onStop()
    smartsign.setTheme(1)
    smartsign.setText("")
end

_enableCallbacks = true
